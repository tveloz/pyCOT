r1: Pact + Np => Pact + Np + Cp; 	# Fijación fotosintética de C, dependiente del N foliar (Rubisco)
r2: Plim + Np + Pp => Pact; 		# Activación de plantas limitadas de nutrientes
r3: Pact => Plim; 			# Transición de activa a limitada de nutrientes
r4: Pact + Cp + Np + Pp => 2Pact; 	# Crecimiento de biomasa vegetal
r5: Pact + Ns => Pact + Np; 		# Absorción directa de N por raíces
r6: Pact + Ps => Pact + Pp; 		# Absorción directa de P por raíces
r7: Pact + Np + Pp => Ns + Ps; 		# Reciclaje de N y P de planta activa al suelo
r8: Plim + Np + Pp => Ns + Ps; 		# Reciclaje de N y P de planta limitada al suelo
r9: M + Ns => M + Nf; 			# Absorción de N por micelio extrarradicular
r10: M + Ps => M + Pf; 			# Absorción de P por micelio extrarradicular
r11: M + Cf + 2Nf + Pf => 2M; 	# Crecimiento de micelio extrarradicular (tejido fúngico rico en N, C:N ~10)
r12: M => Ns + Ps; 			# Mortalidad micelial y reciclaje
r13: F + Cp + Nf => F + Cf + Np; 	# Intercambio de C por N planta-hongo
r14: F + Nf => F + Np; 			# Transferencia de N del hongo a la planta
r15: F + Cp + Pf => F + Cf + Pp; 	# Intercambio de C por P planta-hongo (pago de C desacoplado del N)
r16: F + Cf + 2Nf + Pf => 2F; 	# Crecimiento fúngico intrarradical (tejido fúngico rico en N, C:N ~10)
r17: F + Pp => Pp + Ns + Ps;		# Supresión de la colonización por el estado de P de la planta (regulación de la asignación)
r18: F => Ns + Ps; 			# Mortalidad fúngica y reciclaje
r19:  => Ns; 				# Aporte externo de N al suelo
r20:  => Ps; 				# Aporte externo de P al suelo
r21: 2Pact => Pact; 			# Reacción de control de Pact
r22: 2Plim => Plim; 			# Reacción de control de Plim
r23: 2Cp => Cp; 			# Reacción de control de Cp
r24: 2Np => Np; 			# Reacción de control de Np
r25: 2Pp => Pp; 			# Reacción de control de Pp
r26: 2F => F; 				# Reacción de control de F
r27: 2M => M; 				# Reacción de control de M
r28: 2Cf => Cf; 			# Reacción de control de Cf
r29: 2Nf => Nf; 			# Reacción de control de Nf
r30: 2Pf => Pf; 			# Reacción de control de Pf
r31: 2Ns => Ns; 			# Reacción de control de Ns
r32: 2Ps => Ps;				# Reacción de control de Ps
